        status: 'Guarded',
        decision: 'Wait',
      },
    ];

    function cycleSimulation() {
      const s = steps[idx];
      document.getElementById('sim-step').textContent = s.step;
      document.getElementById('sim-topic').textContent = s.topic;
      document.getElementById('sim-caption').textContent = s.caption;
      document.getElementById('sim-eta').textContent = s.eta;
      document.getElementById('sim-status').innerHTML =
        '<span class="h-1.5 w-1.5 rounded-full bg-emerald-300 animate-pulse"></span>' +
        '<span>' + s.status + '</span>';
      idx = (idx + 1) % steps.length;

    setInterval(cycleSimulation, 5500);

    const roleButtons = document.querySelectorAll('.role-btn');
    let selectedRole = '';
    roleButtons.forEach(btn => {

    let autoEnabled = true;
    autoToggle.addEventListener('click', () => {
      autoEnabled = !autoEnabled;

        autoToggle.classList.remove('bg-slate-700');
        autoToggle.classList.add('bg-emerald-500/30');
